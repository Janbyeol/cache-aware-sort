CXX      := g++
CXXFLAGS := -O3 -march=native -std=c++20 -Wall -Wextra -Iinclude

COMMON_SRCS := \
	src/baseline_sorts.cpp \
	src/idea1_cache_leaf_quick.cpp \
	src/idea2_mean_pivot_quick.cpp \
	src/idea3_boundary_trimmed_merge.cpp \
	src/idea4_tiled_pingpong_merge.cpp \
	src/benchmark_api.cpp

all: test_4b test_8b bench_4b bench_8b

# 1. 유닛 테스트 빌드 (4바이트 키 / 8바이트 레코드)
test_4b: $(COMMON_SRCS) src/test_main.cpp
	$(CXX) $(CXXFLAGS) $^ -o $@

test_8b: $(COMMON_SRCS) src/test_main.cpp
	$(CXX) $(CXXFLAGS) -DBENCH_RECORD=1 $^ -o $@

# 2. 벤치마크 빌드 (4바이트 키 / 8바이트 레코드)
bench_4b: $(COMMON_SRCS) src/benchmark_main.cpp
	$(CXX) $(CXXFLAGS) $^ -o $@

bench_8b: $(COMMON_SRCS) src/benchmark_main.cpp
	$(CXX) $(CXXFLAGS) -DBENCH_RECORD=1 $^ -o $@

# 실행 타깃
test: test_4b test_8b
	./test_4b
	./test_8b

bench: bench_4b bench_8b
	./bench_4b
	./bench_8b

clean:
	rm -f test_4b test_8b bench_4b bench_8b *.exe *.csv

.PHONY: all test bench clean
